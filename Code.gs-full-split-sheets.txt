function doPost(e) {
  // 단어 테스트 비밀번호 확인
  if (e.parameter.action === "checkVocaPassword") {
    return checkVocaPassword_(e);
  }

  if (e.parameter.action === "playWordChain") {
    return playWordChain_(e);
  }

  // 내신/모의고사 테스트 완료 기록 저장
  if (e.parameter.action === "saveTest") {
    return saveTestCompletion_(e);
  }

  // TOEFL 작문 제출 저장
  if (e.parameter.action === "saveWriting") {
    return saveWritingSubmission_(e);
  }

  // 학생이 이름, 문제, 암호로 최근 작문 결과 찾기
  if (e.parameter.action === "findWritingSubmission") {
    return findWritingSubmission_(e);
  }

  // 단어장 성적표 저장 기능
  if (e.parameter.action === "saveScore") {
    var scoreSheet = getScoresSheet_();
    if (scoreSheet) {
      scoreSheet.appendRow([
        new Date(),
        e.parameter.studentName,
        e.parameter.subjectDay,
        e.parameter.range,
        e.parameter.score
      ]);
    }
    sendMiddleVocaResultEmail_(e);
    return ContentService.createTextOutput("성공");
  }

  // 관리자 화면에서 방 만들기 저장 기능
  var passageSheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("Passages");
  if (passageSheet) {
    passageSheet.appendRow([
      e.parameter.grade,
      e.parameter.subject,
      e.parameter.set,
      e.parameter.password,
      "",
      new Date()
    ]);
    CacheService.getScriptCache().remove("writingSetPasswordMap_v1");
  }
  return ContentService.createTextOutput("성공");
}

      var rowPassword = String(rows[i][2] || "");

      if (rowSubject !== subject) continue;

      if (rowDay === day || (gradeKey && rowDay === gradeKey)) {
        exactPassword = rowPassword;
        break;
      }

      if (rowDay === "" && defaultPassword === null) {
        defaultPassword = rowPassword;
      }
    }
  }

  var correctPassword = exactPassword !== null ? exactPassword : defaultPassword;
  var ok = correctPassword !== null && password === correctPassword;

  return ContentService
    .createTextOutput(JSON.stringify({ ok: ok }))
    .setMimeType(ContentService.MimeType.JSON);
}

function getVocaGradeKey_(day) {
  var value = String(day || "").trim();
  var match = value.match(/^(중[1-3]|고[1-3])/);
  return match ? match[1] : "";
}

function getWordChain_(e) {
  var lock = LockService.getScriptLock();
  var hasLock = false;

  try {
    lock.waitLock(25000);
    hasLock = true;

    var ss = SpreadsheetApp.getActiveSpreadsheet();
    ensureWordChainDictionary_(ss, false);
    var state = getOrCreateWordChainState_(ss);
    return wordChainJson_(buildWordChainPayload_(ss, state, normalizeWordChainPlayerName_(e.parameter.studentName)));
  } catch (error) {
    return wordChainJson_({
      ok: false,
      message: "게임 단어를 불러오지 못했습니다. 잠시 후 다시 시도해 주세요."
    });
  } finally {
    if (hasLock) lock.releaseLock();
  }
}

function playWordChain_(e) {
  var lock = LockService.getScriptLock();
  var hasLock = false;

  try {
    lock.waitLock(25000);
    hasLock = true;

    var ss = SpreadsheetApp.getActiveSpreadsheet();
    ensureWordChainDictionary_(ss, false);
    var state = getOrCreateWordChainState_(ss);
    var playerName = normalizeWordChainPlayerName_(e.parameter.studentName);
    var requestedWord = normalizeWordChainWord_(e.parameter.word);

    if (!playerName) {
      return wordChainFailure_(ss, state, playerName, "이름을 다시 입력해 주세요.");
    }

    if (!requestedWord) {
      return wordChainFailure_(ss, state, playerName, "영어 단어만 입력해 주세요.");
    }

    if (requestedWord.charAt(0) !== state.requiredLetter) {
      return wordChainFailure_(
        ss,
        state,
        playerName,
        "다른 학생이 방금 단어를 이어 썼습니다. 지금은 " + state.requiredLetter.toUpperCase() + "로 시작하는 단어 차례입니다."
      );
    }

    var entry = getWordChainEntry_(ss, requestedWord);
    if (!entry) {
      return wordChainFailure_(ss, state, playerName, "TOEFL·TOEIC 단어장에 있는 단어로 입력해 주세요.");
    }

    var usedWords = parseWordChainUsedWords_(state.usedWords);
    if (usedWords[entry.word]) {
      return wordChainFailure_(ss, state, playerName, "오늘 이미 나온 단어입니다. 다른 단어를 골라 주세요.");
    }

    var nextUsedWords = wordChainUsedWordsWith_(state.usedWords, entry.word);
    var nextUsedMap = parseWordChainUsedWords_(nextUsedWords);
    var nextCandidates = getWordChainAvailableEntries_(ss, entry.lastLetter, nextUsedMap);
    if (!nextCandidates.length) {
      return wordChainFailure_(
        ss,
        state,
        playerName,
        "그 단어 뒤에는 이어갈 단어가 없습니다. 다른 단어를 골라 주세요."
      );
    }

    var nextTotal = Number(state.totalWords || 0) + 1;
    var nextPlayerCounts = incrementWordChainPlayerCount_(state.playerCounts, playerName);
    var now = new Date();
    state.sheet.getRange(state.rowNumber, 1, 1, 10).setValues([[
      state.dateKey,
      state.seedWord,
      entry.word,
      entry.meaning,
      entry.lastLetter,
      nextTotal,
      now,
      playerName,
      nextUsedWords,
      nextPlayerCounts
    ]]);

    getWordChainEntriesSheet_(ss).appendRow([
      state.dateKey,
      nextTotal,
      now,
      playerName,
      entry.word,
      entry.meaning,
      String(e.parameter.hintUsed || "") === "TRUE",
      entry.source
    ]);

    state.currentWord = entry.word;
    state.currentMeaning = entry.meaning;
    state.requiredLetter = entry.lastLetter;
    state.totalWords = nextTotal;
    state.updatedAt = now;
    state.lastPlayer = playerName;
    state.usedWords = nextUsedWords;
    state.playerCounts = nextPlayerCounts;

    var payload = buildWordChainPayload_(ss, state, playerName);
    payload.acceptedWord = entry.word;
    payload.acceptedMeaning = entry.meaning;
    payload.relatedWords = getWordChainRelatedWords_(ss, entry.firstLetter, entry.word);
    return wordChainJson_(payload);
  } catch (error) {
    return wordChainJson_({
      ok: false,
      message: "등록하지 못했습니다. 잠시 후 다시 시도해 주세요."
    });
  } finally {
    if (hasLock) lock.releaseLock();
  }
}

function wordChainFailure_(ss, state, playerName, message) {
  var payload = buildWordChainPayload_(ss, state, playerName);
  payload.ok = false;
  payload.message = message;
  return wordChainJson_(payload);
}

function ensureWordChainDictionary_(ss, forceRefresh) {
  var properties = PropertiesService.getScriptProperties();
  var sourceSignature = getWordChainSourceSignature_(ss);
  var sheet = ss.getSheetByName(WORD_CHAIN_WORDS_SHEET);

  if (!forceRefresh && sheet && sheet.getLastRow() > 1 && properties.getProperty("wordChainSourceSignature") === sourceSignature) {
    return sheet;
  }

  var uniqueWords = {};
  WORD_CHAIN_SOURCE_SHEETS.forEach(function(sourceName) {
    var sourceSheet = ss.getSheetByName(sourceName);
    if (!sourceSheet || sourceSheet.getLastRow() < 2) return;

    var rows = sourceSheet.getDataRange().getValues();
    for (var i = 1; i < rows.length; i++) {
      var word = normalizeWordChainWord_(rows[i][4]);
      var meaning = String(rows[i][3] || "").trim();
      if (!word || word.length < 4 || !meaning || uniqueWords.hasOwnProperty(word)) continue;

      uniqueWords[word] = {
        source: sourceName,
        day: String(rows[i][1] || ""),
        word: word,
        meaning: meaning,
        firstLetter: word.charAt(0),
        lastLetter: word.charAt(word.length - 1)
      };
    }
  });

  var entries = [];
  Object.keys(uniqueWords).forEach(function(word) {
    entries.push(uniqueWords[word]);
  });
  entries = keepWordChainSafeEntries_(entries);

  if (!entries.length) {
    throw new Error("No usable TOEFL or TOEIC word-chain entries");
  }

  if (!sheet) sheet = ss.insertSheet(WORD_CHAIN_WORDS_SHEET);
  sheet.clearContents();
  sheet.getRange(1, 1, 1, 6).setValues([[
    "source", "day", "word", "meaning", "firstLetter", "lastLetter"
  ]]);
  sheet.getRange(2, 1, entries.length, 6).setValues(entries.map(function(entry) {
    return [entry.source, entry.day, entry.word, entry.meaning, entry.firstLetter, entry.lastLetter];
  }));

  clearWordChainLetterCache_();
  properties.setProperty("wordChainSourceSignature", sourceSignature);
  return sheet;
}

function getWordChainSourceSignature_(ss) {
  return WORD_CHAIN_SOURCE_SHEETS.map(function(sourceName) {
    var sheet = ss.getSheetByName(sourceName);
    return sourceName + ":" + (sheet ? sheet.getLastRow() : 0);
  }).join("|") + "|rules:" + WORD_CHAIN_DICTIONARY_VERSION;
}

function keepWordChainSafeEntries_(entries) {
  var current = entries.slice();
  var changed = true;

  while (changed) {
    var startCounts = {};
    current.forEach(function(entry) {
      startCounts[entry.firstLetter] = Number(startCounts[entry.firstLetter] || 0) + 1;
    });

    var next = current.filter(function(entry) {
      return Number(startCounts[entry.lastLetter] || 0) > 0;
    });
    changed = next.length !== current.length;
    current = next;
  }

  return current;
}

function clearWordChainLetterCache_() {
  var cache = CacheService.getScriptCache();
  for (var code = 97; code <= 122; code++) {
    cache.remove("wordChainLetter_" + String.fromCharCode(code));
  }
}

function getWordChainEntriesForLetter_(ss, firstLetter) {
  var letter = String(firstLetter || "").toLowerCase();
  if (!/^[a-z]$/.test(letter)) return [];

  var cache = CacheService.getScriptCache();
  var cacheKey = "wordChainLetter_" + letter;
  var cached = cache.get(cacheKey);
  if (cached) {
    try {
      return JSON.parse(cached);
    } catch (error) {
      cache.remove(cacheKey);
    }
  }

  var sheet = getWordChainWordsSheet_(ss);
  if (!sheet || sheet.getLastRow() < 2) return [];

  var rows = sheet.getRange(2, 1, sheet.getLastRow() - 1, 6).getValues();
  var entries = rows.filter(function(row) {
    return String(row[4] || "").toLowerCase() === letter;
  }).map(function(row) {
    return {
      source: String(row[0] || ""),
      day: String(row[1] || ""),
      word: String(row[2] || ""),
      meaning: String(row[3] || ""),
      firstLetter: String(row[4] || ""),
      lastLetter: String(row[5] || "")
    };
  });

  var serialized = JSON.stringify(entries);
  if (serialized.length < 90000) cache.put(cacheKey, serialized, 1800);
  return entries;
}

function getWordChainEntry_(ss, word) {
  var entries = getWordChainEntriesForLetter_(ss, word.charAt(0));
  for (var i = 0; i < entries.length; i++) {
    if (entries[i].word === word) return entries[i];
  }
  return null;
}

function getWordChainAvailableEntries_(ss, firstLetter, usedWords) {
  return getWordChainEntriesForLetter_(ss, firstLetter).filter(function(entry) {
    return !usedWords[entry.word];
  });
}

function getOrCreateWordChainState_(ss) {
  var sheet = getWordChainStateSheet_(ss);
  var dateKey = getWordChainDateKey_();
  var state = findWordChainState_(sheet, dateKey);
  if (state) return state;

  var seed = getWordChainSeed_(ss);
  var rowNumber = sheet.getLastRow() + 1;
  var now = new Date();
  var values = [[
    dateKey,
    seed.word,
    seed.word,
    seed.meaning,
    seed.lastLetter,
    0,
    now,
    "",
    wordChainUsedWordsWith_("", seed.word),
    "[]"
  ]];
  sheet.getRange(rowNumber, 1, 1, 10).setValues(values);
  return wordChainStateFromRow_(sheet, rowNumber, values[0]);
}

function findWordChainState_(sheet, dateKey) {
  if (sheet.getLastRow() < 2) return null;
  var match = sheet.getRange(2, 1, sheet.getLastRow() - 1, 1)
    .createTextFinder(dateKey)
    .matchEntireCell(true)
    .findNext();
  if (!match) return null;
  var rowNumber = match.getRow();
  return wordChainStateFromRow_(sheet, rowNumber, sheet.getRange(rowNumber, 1, 1, 10).getValues()[0]);
}

function getWordChainSeed_(ss) {
  var sheet = getWordChainWordsSheet_(ss);
  if (!sheet || sheet.getLastRow() < 2) throw new Error("Word-chain dictionary is empty");

  var rows = sheet.getRange(2, 1, sheet.getLastRow() - 1, 6).getValues();
  var entries = rows.map(function(row) {
    return {
      source: String(row[0] || ""),
      day: String(row[1] || ""),
      word: String(row[2] || ""),
      meaning: String(row[3] || ""),
      firstLetter: String(row[4] || ""),
      lastLetter: String(row[5] || "")
    };
  });
  var startCounts = {};
  entries.forEach(function(entry) {
    startCounts[entry.firstLetter] = Number(startCounts[entry.firstLetter] || 0) + 1;
  });
  var seedEntries = entries.filter(function(entry) {
    var nextCount = Number(startCounts[entry.lastLetter] || 0);
    if (entry.firstLetter === entry.lastLetter) nextCount--;
    return nextCount >= WORD_CHAIN_HINT_COUNT;
  });
  if (!seedEntries.length) throw new Error("No safe word-chain seed");

  var seed = seedEntries[Math.floor(Math.random() * seedEntries.length)];
  return {
    source: seed.source,
    day: seed.day,
    word: seed.word,
    meaning: seed.meaning,
    firstLetter: seed.firstLetter,
    lastLetter: seed.lastLetter
  };
}

function buildWordChainPayload_(ss, state, playerName) {
  var usedWords = parseWordChainUsedWords_(state.usedWords);
  return {
    ok: true,
    dateKey: state.dateKey,
    currentWord: state.currentWord,
    currentMeaning: state.currentMeaning,
    requiredLetter: state.requiredLetter,
    totalCount: Number(state.totalWords || 0),
    myCount: getWordChainPlayerCount_(state.playerCounts, playerName),
    lastPlayerName: state.lastPlayer ? maskStudentName_(state.lastPlayer) : "",
    hints: getWordChainHints_(ss, state.requiredLetter, usedWords),
    recentEntries: getWordChainRecentEntries_(ss, state.dateKey, 10)
  };
}

function getWordChainHints_(ss, requiredLetter, usedWords) {
  return pickWordChainEntries_(
    getWordChainAvailableEntries_(ss, requiredLetter, usedWords),
    WORD_CHAIN_HINT_COUNT
  ).map(function(entry) {
    return entry.meaning;
  });
}

function getWordChainRelatedWords_(ss, firstLetter, acceptedWord) {
  var entries = getWordChainEntriesForLetter_(ss, firstLetter).filter(function(entry) {
    return entry.word !== acceptedWord;
  });
  return pickWordChainEntries_(entries, 3).map(function(entry) {
    return { word: entry.word, meaning: entry.meaning };
  });
}

function pickWordChainEntries_(entries, limit) {
  var pool = entries.slice();
  for (var i = pool.length - 1; i > 0; i--) {
    var randomIndex = Math.floor(Math.random() * (i + 1));
    var temp = pool[i];
    pool[i] = pool[randomIndex];
    pool[randomIndex] = temp;
  }
  return pool.slice(0, limit);
}

function getWordChainRecentEntries_(ss, dateKey, limit) {
  var sheet = getWordChainEntriesSheet_(ss);
  if (sheet.getLastRow() < 2) return [];

  var startRow = Math.max(2, sheet.getLastRow() - 499);
  var rows = sheet.getRange(startRow, 1, sheet.getLastRow() - startRow + 1, 8).getValues();
  var entries = [];
  for (var i = rows.length - 1; i >= 0 && entries.length < limit; i--) {
    if (wordChainDateKeyFromValue_(rows[i][0]) !== dateKey) continue;
    entries.push({
      playerName: maskStudentName_(rows[i][3]),
      word: String(rows[i][4] || ""),
      meaning: String(rows[i][5] || "")
    });
  }
  return entries;
}

function normalizeWordChainWord_(value) {
  var word = String(value || "").trim();
  if (!/^[A-Za-z]+$/.test(word)) return "";
  return word.toLowerCase();
}

function normalizeWordChainPlayerName_(value) {
  return String(value || "")
    .replace(/[\r\n\t]+/g, " ")
    .replace(/\s+/g, " ")
    .replace(/[<>]/g, "")
    .trim()
    .slice(0, 20);
}

function parseWordChainUsedWords_(value) {
  var result = {};
  String(value || "").split("|").forEach(function(word) {
    if (word) result[word] = true;
  });
  return result;
}

function wordChainUsedWordsWith_(value, word) {
  var used = String(value || "");
  var token = "|" + word + "|";
  if (used.indexOf(token) !== -1) return used;
  return used + token;
}

function getWordChainPlayerCount_(serializedCounts, playerName) {
  if (!playerName) return 0;
  var counts = parseWordChainPlayerCounts_(serializedCounts);
  for (var i = 0; i < counts.length; i++) {
    if (counts[i].name === playerName) return Number(counts[i].count || 0);
  }
  return 0;
}

function incrementWordChainPlayerCount_(serializedCounts, playerName) {
  var counts = parseWordChainPlayerCounts_(serializedCounts);
  for (var i = 0; i < counts.length; i++) {
    if (counts[i].name === playerName) {
      counts[i].count = Number(counts[i].count || 0) + 1;
      return JSON.stringify(counts);
    }
  }
  counts.push({ name: playerName, count: 1 });
  return JSON.stringify(counts);
}

function parseWordChainPlayerCounts_(value) {
  try {
    var counts = JSON.parse(String(value || "[]"));
    return Array.isArray(counts) ? counts : [];
  } catch (error) {
    return [];
  }
}

function formatWritingSubmittedAt_(value) {
  if (!(value instanceof Date)) return String(value || "");
  var timeZone = Session.getScriptTimeZone() || "Asia/Tokyo";
  return Utilities.formatDate(value, timeZone, "yyyy.MM.dd HH:mm");
}

function getWritingResultSets_() {
  return writingResultJson_({
    sets: Object.keys(getWritingSetPasswordMap_()).sort()
  });
}

function getWritingSetPasswordMap_() {
  var cache = CacheService.getScriptCache();
  var cacheKey = "writingSetPasswordMap_v1";
  var cached = cache.get(cacheKey);

  if (cached) {
    try {
      return JSON.parse(cached);
    } catch (error) {
      cache.remove(cacheKey);
    }
  }

  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName("Passages");
  var setMap = {};

  if (sheet && sheet.getLastRow() > 1) {
    var rows = sheet.getRange(2, 1, sheet.getLastRow() - 1, 4).getValues();
    rows.forEach(function(row) {
      var grade = normalizeGradeName_(row[0]);
      var subject = String(row[1] || "").trim();
      var problemSet = String(row[2] || "").trim();
      if (grade === "TOEFL" && subject === "작문연습" && problemSet && !setMap.hasOwnProperty(problemSet)) {
        setMap[problemSet] = String(row[3] || "");
      }
    });
  }

  cache.put(cacheKey, JSON.stringify(setMap), 30);
  return setMap;
}

function findWritingSubmission_(e) {
  var studentName = String(e.parameter.studentName || "").trim();
  var problemSet = String(e.parameter.problemSet || "").trim();
  var password = String(e.parameter.password || "");

  if (!studentName || !problemSet || !password || !isWritingSetPasswordValid_(problemSet, password)) {
    return writingResultJson_({ found: false });
  }

  var sheet = getWritingSubmissionsSheet_();
  var lastRow = sheet.getLastRow();
  if (lastRow < 2) return writingResultJson_({ found: false });

  // 답안, 첨삭 프롬프트, 첨삭문처럼 긴 열은 결과를 실제로 열 때만 읽는다.
  var lookupRows = sheet.getRange(2, 2, lastRow - 1, 4).getValues();
  var submissionIds = sheet.getRange(2, 14, lastRow - 1, 1).getValues();
  for (var i = lookupRows.length - 1; i >= 0; i--) {
    if (String(lookupRows[i][0] || "").trim() !== studentName) continue;
    if (String(lookupRows[i][3] || "").trim() !== problemSet) continue;

    var submissionId = String(submissionIds[i][0] || "").trim();
    if (submissionId) {
      return writingResultJson_({
        found: true,
        submissionId: submissionId
      });
    }
  }

  return writingResultJson_({ found: false });
}

function isWritingSetPasswordValid_(problemSet, password) {
  var setMap = getWritingSetPasswordMap_();
  return setMap.hasOwnProperty(problemSet) && setMap[problemSet] === password;
}

function writingResultJson_(data) {
  return ContentService
    .createTextOutput(JSON.stringify(data))
    .setMimeType(ContentService.MimeType.JSON);
}


function doGet(e) {
  if (e.parameter.action === "getWordChain") {
    return getWordChain_(e);
  }

  // 작문 결과 찾기 화면의 문제세트 목록
  if (e.parameter.action === "getWritingResultSets") {
    return getWritingResultSets_(e);
  }

  // 작문 제출 결과: 개인 제출번호로 원문과 첨삭만 조회
  if (e.parameter.action === "getWritingSubmission") {
    return getWritingSubmission_(e);
  }

  // 공개용 단어 결과 보기: Scores 시트에서 필요한 항목만 보내기
  if (e.parameter.action === "getPublicScoreResults") {
    return getPublicScoreResults_(e);
  }

  // 공개용 학생 결과 보기: test 시트에서 필요한 항목만 보내기
  if (e.parameter.action === "getPublicTestResults") {
    return getPublicTestResults_(e);
  }

  // 단어: 과목별로 나눈 시트만 읽기
  if (e.parameter.action === "getVoca") {
    var targetSubject = String(e.parameter.subject || "").trim();
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var vocaSheetName = getVocaSheetName_(targetSubject);
    var vocaSheet = ss.getSheetByName(vocaSheetName) || ss.getSheetByName("Voca");
    var usingFallbackVocaSheet = vocaSheet && vocaSheet.getName() === "Voca";
    var vocaResult = [];

    if (vocaSheet) {
      var vocaRows = vocaSheet.getDataRange().getValues();
      for (var i = 1; i < vocaRows.length; i++) {
        var rowSubject = String(vocaRows[i][0] || "").trim();
        var rowDay = vocaRows[i][1];
        if (rowDay && (!targetSubject || !usingFallbackVocaSheet || rowSubject === targetSubject)) {
          vocaResult.push({
            subject: String(vocaRows[i][0] || targetSubject),
            day: String(vocaRows[i][1]),
            num: Number(vocaRows[i][2]),
            question: String(vocaRows[i][3]),
            answer: String(vocaRows[i][4]),
            w1: String(vocaRows[i][5]),
            w2: String(vocaRows[i][6]),
            w3: String(vocaRows[i][7])
          });
        }
      }
    }

    return ContentService
      .createTextOutput(JSON.stringify(vocaResult))
      .setMimeType(ContentService.MimeType.JSON);
  }

  // 문제 풀이: 학년별로 나눈 Questions 시트만 읽기
  if (e.parameter.action === "getQuiz") {
    var targetSet = String(e.parameter.set || "");
    var targetGrade = normalizeGradeName_(e.parameter.grade);
    var result = { passage: "", questions: [] };
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var cache = CacheService.getScriptCache();
    var cacheKey = "quiz:" + targetGrade + ":" + targetSet;
    var cachedQuiz = cache.get(cacheKey);

    if (cachedQuiz) {
      return ContentService
        .createTextOutput(cachedQuiz)
        .setMimeType(ContentService.MimeType.JSON);
    }

    result.passage = getQuizPassage_(ss, targetSet, targetGrade);
    result.questions = getQuizQuestions_(ss, targetSet, targetGrade);

    var quizJson = JSON.stringify(result);
    if (quizJson.length < 90000) {
      cache.put(cacheKey, quizJson, 300);
    }

    return ContentService
      .createTextOutput(quizJson)
      .setMimeType(ContentService.MimeType.JSON);
  }

  // 첫 화면 목록은 Passages에서 가져오기
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("Passages");
  var dataList = [];
  if (sheet) {
    var rows = sheet.getDataRange().getValues();
    for (var j = 1; j < rows.length; j++) {
      var row = rows[j];
      if (row[0] && row[1] && row[2]) {
        dataList.push({
          grade: normalizeGradeName_(row[0]),
          subject: String(row[1]),
          set: String(row[2]),
          password: String(row[3])
        });
      }
    }
  }

  return ContentService
    .createTextOutput(JSON.stringify(dataList))
    .setMimeType(ContentService.MimeType.JSON);
}

function getPublicScoreResults_(e) {
  var sheet = getScoresSheet_();
  var result = [];
  var limit = Math.min(Number(e.parameter.limit || 100), 200);

  if (sheet && sheet.getLastRow() > 1) {
    var rows = sheet.getDataRange().getValues();
    for (var i = rows.length - 1; i >= 1 && result.length < limit; i--) {
      var row = rows[i];
      if (!row[0] && !row[1]) continue;

      result.push({
        date: formatPublicDate_(row[0]),
        studentName: maskStudentName_(row[1]),
        subjectDay: String(row[2] || ""),
        range: String(row[3] || ""),
        score: String(row[4] || "")
      });
    }
  }

  return ContentService
    .createTextOutput(JSON.stringify({
      type: "publicScoreResults",
      rows: result
    }))
    .setMimeType(ContentService.MimeType.JSON);
}

function formatPublicDate_(value) {
  if (!value) return "";
  var date = value instanceof Date ? value : new Date(value);
  if (isNaN(date.getTime())) return String(value);
  return Utilities.formatDate(date, Session.getScriptTimeZone(), "yyyy-MM-dd");
}

function maskStudentName_(value) {
  var name = String(value || "").trim();
  if (!name) return "";
  if (name.indexOf("즌") === 0 || name.indexOf("즈") === 0) return "Z**";
  var chars = Array.from(name);
  return getNameInitial_(chars[0]) + "**";
}

function getVocaSheetName_(subject) {
  if (subject === "토플") return "Voca_토플";
  if (subject === "토익") return "Voca_토익";
  if (subject === "SAT") return "Voca_SAT";
  if (subject === "교과서_중등") return "Voca_교과서_중등";
  if (subject === "교과서_고등") return "Voca_교과서_고등";
  if (subject === "교과서") return "Voca_교과서";
  return "Voca";
}

function getQuestionSheetName_(grade) {
  grade = normalizeGradeName_(grade);
  if (grade === "중1") return "Questions_중1";
  if (grade === "중2") return "Questions_중2";
  if (grade === "중3") return "Questions_중3";
  if (grade === "고1") return "Questions_고1";
  if (grade === "고2") return "Questions_고2";
  if (grade === "고3") return "Questions_고3";
  if (grade === "TOEFL") return "Questions_TOEFL";
  if (grade === "SAT") return "Questions_SAT";
  return "Questions";
}

function normalizeGradeName_(grade) {
  var value = String(grade || "").trim();
  var upper = value.toUpperCase();
  if (value === "토플" || upper === "TOEFL") return "TOEFL";
  if (value === "사트" || upper === "SAT") return "SAT";
  return value;
}

function getQuizPassage_(ss, targetSet, targetGrade) {
  var passageSheet = ss.getSheetByName("Passages");
  if (!passageSheet || !targetSet) return "";

  var lastRow = passageSheet.getLastRow();
  if (lastRow < 2) return "";

  var setMatches = passageSheet
    .getRange(2, 3, lastRow - 1, 1)
    .createTextFinder(targetSet)
    .matchEntireCell(true)
    .findAll();

  for (var i = 0; i < setMatches.length; i++) {
    var rowNumber = setMatches[i].getRow();
    var row = passageSheet.getRange(rowNumber, 1, 1, 5).getValues()[0];
    var rowGrade = normalizeGradeName_(row[0]);

    if (targetGrade && rowGrade !== targetGrade) continue;

    var passageCell = passageSheet.getRange(rowNumber, 5);
    return resolvePassageText_(
      ss,
      row[4],
      passageCell.getRichTextValue(),
      passageCell.getBackground()
    );
  }

  return "";
}

function getQuizQuestions_(ss, targetSet, targetGrade) {
  var questionSheetName = getQuestionSheetName_(targetGrade);
  var qSheet = ss.getSheetByName(questionSheetName) || ss.getSheetByName("Questions");
  var questions = [];

  if (!qSheet || !targetSet) return questions;

  var lastRow = qSheet.getLastRow();
  if (lastRow < 2) return questions;

  var setMatches = qSheet
    .getRange(2, 1, lastRow - 1, 1)
    .createTextFinder(targetSet)
    .matchEntireCell(true)
    .findAll()
    .sort(function(a, b) {
      return a.getRow() - b.getRow();
    });

  setMatches.forEach(function(match) {
    var row = qSheet.getRange(match.getRow(), 1, 1, 9).getValues()[0];
    questions.push({
      question: String(row[1]),
      c1: String(row[2]),
      c2: String(row[3]),
      c3: String(row[4]),
      c4: String(row[5]),
      answer: String(row[6]),
      explanation: String(row[7] || ""),
      type: String(row[8] || ""),
      blankSentence: String(row[2] || ""),
      wordBank: String(row[3] || "")
    });
  });

  return questions;
}

function resolvePassageText_(ss, passageValue, passageRichTextValue, passageBackground) {
  var value = String(passageValue || "").trim();
  if (!value) return "";

  var passageTextSheet = ss.getSheetByName("PassageTexts");
  if (!passageTextSheet) return richTextToHtml_(passageRichTextValue, passageBackground) || value;

  var lastRow = passageTextSheet.getLastRow();
  if (lastRow > 1) {
    var match = passageTextSheet
      .getRange(2, 1, lastRow - 1, 1)
      .createTextFinder(value)
      .matchEntireCell(true)
      .findNext();

    if (match) {
      var textCell = passageTextSheet.getRange(match.getRow(), 2);
      return richTextToHtml_(textCell.getRichTextValue(), textCell.getBackground()) || textCell.getValue() || "";
    }
  }

  // 기존 Passages 데이터처럼 E열에 지문 본문이 직접 들어 있는 경우도 계속 지원
  return richTextToHtml_(passageRichTextValue, passageBackground) || value;
}

function richTextToHtml_(richTextValue, backgroundColor) {
  if (!richTextValue) return "";

  var runs = richTextValue.getRuns();
  if (!runs || runs.length === 0) {
    return wrapHighlight_(escapeHtml_(richTextValue.getText()), backgroundColor);
  }

  var html = "";
  runs.forEach(function(run) {
    var text = escapeHtml_(run.getText());
    var style = run.getTextStyle();
    if (style && style.isUnderline()) {
      text = "<u>" + text + "</u>";
    }

    if (isHighlightedBackground_(backgroundColor)) {
      text = "<mark>" + text + "</mark>";
    }

    html += text;
  });

  return html;
}

function wrapHighlight_(html, backgroundColor) {
  if (isHighlightedBackground_(backgroundColor)) {
    return "<mark>" + html + "</mark>";
  }
  return html;
}

function isHighlightedBackground_(backgroundColor) {
  var color = String(backgroundColor || "").toLowerCase();
  return color && color !== "#ffffff" && color !== "white" && color !== "#fff";
}

function escapeHtml_(text) {
  return String(text || "")
    .replace(/&/g, "&amp;")
    .replace(/</g, "&lt;")
    .replace(/>/g, "&gt;");
}

function isMiddleGrade_(grade) {

}
